cmake_minimum_required(VERSION 3.22)
project(UniversalDevKitXamlTests LANGUAGES CXX)
get_filename_component(udkRoot "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(UniversalDevKit_COMPONENTS "XamlRuntime" CACHE STRING "Components" FORCE)
set(UniversalDevKitXamlLinkage "Static" CACHE STRING "XAML component linkage: Static or Dynamic")
if(NOT UniversalDevKitXamlLinkage MATCHES "^(Static|Dynamic)$")
    message(FATAL_ERROR "Select Static or Dynamic XAML linkage")
endif()
set(UniversalDevKit_BUILD_STATIC OFF CACHE BOOL "Static" FORCE)
set(UniversalDevKit_BUILD_DYNAMIC OFF CACHE BOOL "Dynamic" FORCE)
if(UniversalDevKitXamlLinkage STREQUAL "Static")
    set(UniversalDevKit_BUILD_STATIC ON CACHE BOOL "Static" FORCE)
else()
    set(UniversalDevKit_BUILD_DYNAMIC ON CACHE BOOL "Dynamic" FORCE)
endif()
set(UniversalDevKit_INSTALL OFF CACHE BOOL "Install" FORCE)
option(UniversalDevKitXaml_BUILD_TESTS "Build XAML tests" ON)
add_subdirectory("${udkRoot}" "${CMAKE_CURRENT_BINARY_DIR}/UDK")
add_subdirectory("${udkRoot}/Projects/Cmake/XamlCompiler" "${CMAKE_CURRENT_BINARY_DIR}/XamlCompiler")
if(UniversalDevKitXaml_BUILD_TESTS)
    if(NOT TARGET udk_xaml_compiler)
        message(FATAL_ERROR "The color animation test requires the host compiler")
    endif()
    enable_testing()
    set(testMarkup "${udkRoot}/Tests/XamlRuntime/ColorAnimationPage.xaml")
    set(testGenerated "${CMAKE_CURRENT_BINARY_DIR}/generated/ColorAnimationPage.xaml.cpp")
    add_custom_command(OUTPUT "${testGenerated}" "${CMAKE_CURRENT_BINARY_DIR}/generated/ColorAnimationPage.xaml.h"
        COMMAND $<TARGET_FILE:udk_xaml_compiler> "${testMarkup}" "${testGenerated}"
        DEPENDS udk_xaml_compiler "${testMarkup}" VERBATIM)
    add_executable(udk_xaml_color_animation_tests "${udkRoot}/Tests/XamlRuntime/ColorAnimationTests.cpp" "${testGenerated}")
    target_include_directories(udk_xaml_color_animation_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(udk_xaml_color_animation_tests PRIVATE UniversalDevKit::XamlRuntime.${UniversalDevKitXamlLinkage})
    add_custom_command(TARGET udk_xaml_color_animation_tests POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:universaldevkit_diagnostic_dynamic> $<TARGET_FILE_DIR:udk_xaml_color_animation_tests>)
    if(MSVC)
        target_compile_options(udk_xaml_color_animation_tests PRIVATE /utf-8 /EHsc)
    endif()
    add_test(NAME XamlRuntime.ColorAnimation COMMAND udk_xaml_color_animation_tests "${testMarkup}")
    if(UniversalDevKitXamlLinkage STREQUAL "Dynamic")
        foreach(target udk_xaml_color_animation_tests)
            add_custom_command(TARGET ${target} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:universaldevkit_xamlruntime_dynamic> $<TARGET_FILE_DIR:${target}>)
        endforeach()
    endif()
endif()

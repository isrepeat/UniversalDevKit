cmake_minimum_required(VERSION 3.22)
project(UniversalDevKitAndroidConsumer LANGUAGES CXX)
string(TOLOWER "udk_${UDK_COMPONENT}_${UDK_MODE}" package_name)
find_package(${package_name} CONFIG REQUIRED)
set(prefab_target "${package_name}::${UDK_COMPONENT}")
get_target_property(header_roots ${prefab_target} INTERFACE_INCLUDE_DIRECTORIES)
foreach(directory IN LISTS header_roots)
    if(EXISTS "${directory}/UniversalDevKit/Android/Consumer.cmake")
        include("${directory}/UniversalDevKit/Android/Consumer.cmake")
    endif()
endforeach()
add_library(udk_android_consumer SHARED Main.cpp)
if(UDK_COMPONENT STREQUAL Diagnostic)
    target_compile_definitions(udk_android_consumer PRIVATE UDK_TEST_DIAGNOSTIC=1)
endif()
set_target_properties(udk_android_consumer PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(UDK_COMPONENT STREQUAL HeleprsTest)
    target_compile_definitions(udk_android_consumer PRIVATE UDK_TEST_HELEPRSTEST=1)
endif()
if(UDK_COMPONENT STREQUAL Helpers)
    target_compile_definitions(udk_android_consumer PRIVATE UDK_TEST_HELPERS=1)
endif()
udk_android_link(udk_android_consumer ${prefab_target})
if(UDK_CONFLICT)
    find_package(udk_heleprstest_dynamic CONFIG REQUIRED)
    udk_android_link(udk_android_consumer udk_heleprstest_dynamic::HeleprsTest)
endif()

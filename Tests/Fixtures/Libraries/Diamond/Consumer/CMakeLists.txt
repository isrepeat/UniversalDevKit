cmake_minimum_required(VERSION 3.22)
project(DiamondConsumers LANGUAGES CXX)
set(DIAMOND_MODE Static CACHE STRING "Static or Dynamic")
option(DIAMOND_PRIVATE_STATIC "Private static dependency variant" OFF)
option(DIAMOND_PUBLIC_STATIC "Public static dependency inside dynamic root" OFF)
option(DIAMOND_CONFLICT "Request incompatible variants" OFF)
if(DIAMOND_CHECKOUT)
    add_subdirectory("${DIAMOND_CHECKOUT}" udk)
else()
    find_package(UniversalDevKit CONFIG REQUIRED COMPONENTS A C)
endif()
foreach(app IN ITEMS App1 App2)
    add_executable(${app} "${app}/Main.cpp")
    set(component A)
    if(app STREQUAL App2)
        set(component C)
    endif()
    target_link_libraries(${app} PRIVATE UniversalDevKit::${component}.${DIAMOND_MODE})
    target_compile_definitions(${app} PRIVATE DIAMOND_TEST_BOUNDARIES=1)
    if(DIAMOND_CHECKOUT)
        target_compile_definitions(${app} PRIVATE DIAMOND_TEST_HEADER_ISOLATION=1)
    endif()
    if(app STREQUAL App1)
        if(DIAMOND_PRIVATE_STATIC)
            target_compile_definitions(${app} PRIVATE DIAMOND_TEST_PUBLIC_B=0)
        else()
            target_compile_definitions(${app} PRIVATE DIAMOND_TEST_PUBLIC_B=1)
        endif()
        if(DIAMOND_MODE STREQUAL Dynamic)
            target_compile_definitions(${app} PRIVATE DIAMOND_TEST_DYNAMIC=1)
        endif()
        if(DIAMOND_PUBLIC_STATIC)
            target_compile_definitions(${app} PRIVATE DIAMOND_TEST_PUBLIC_STATIC=1)
        endif()
    endif()
    set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${app}")
    if(DIAMOND_MODE STREQUAL Dynamic)
        add_custom_command(TARGET ${app} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:${app}> $<TARGET_FILE_DIR:${app}>
            COMMAND_EXPAND_LISTS
        )
    endif()
endforeach()
if(DIAMOND_CONFLICT)
    if(DIAMOND_MODE STREQUAL Dynamic)
        target_link_libraries(App1 PRIVATE UniversalDevKit::C.Static)
    else()
        target_link_libraries(App1 PRIVATE UniversalDevKit::C.Dynamic)
    endif()
endif()

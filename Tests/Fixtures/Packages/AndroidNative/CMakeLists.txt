cmake_minimum_required(VERSION 3.22)
project(UdkPackageDeliveryAndroidNative LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Android toolchain is required")
endif()
add_library(udk_packagedelivery STATIC Probe.cpp)
target_compile_features(udk_packagedelivery PUBLIC cxx_std_20)
set_target_properties(udk_packagedelivery PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS udk_packagedelivery ARCHIVE DESTINATION "lib/${ANDROID_ABI}")
install(FILES Probe.h DESTINATION include)

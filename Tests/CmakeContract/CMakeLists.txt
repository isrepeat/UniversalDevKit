cmake_minimum_required(VERSION 3.22)
project(PackageContract LANGUAGES CXX)
set(TEST_COMPONENT Helpers CACHE STRING "Required component")
set(TEST_TARGET Helpers.Static CACHE STRING "Required variant")
set(TEST_CONFLICT "" CACHE STRING "Additional conflicting variant")
set(TEST_CHECK Missing CACHE STRING "Test scenario")
if(TEST_CHECK STREQUAL "SourceConflict")
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." udk)
else()
    find_package(UniversalDevKit 0.1 CONFIG REQUIRED COMPONENTS ${TEST_COMPONENT})
endif()

add_executable(contract Main.cpp)
target_link_libraries(contract PRIVATE UniversalDevKit::${TEST_TARGET})
if(TEST_CONFLICT)
    if(TEST_CHECK STREQUAL "DirectConflict")
        target_link_libraries(contract PRIVATE UniversalDevKit::${TEST_CONFLICT})
    else()
        add_library(intermediate INTERFACE)
        target_link_libraries(intermediate INTERFACE UniversalDevKit::${TEST_CONFLICT})
        target_link_libraries(contract PRIVATE intermediate)
    endif()
endif()

cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(UniversalDevKitXamlCompiler LANGUAGES CXX)
    get_filename_component(udkRoot "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
    file(TO_CMAKE_PATH "${udkRoot}/!artifacts/" artifactPrefix)
    file(TO_CMAKE_PATH "${CMAKE_BINARY_DIR}/" buildPrefix)
    string(FIND "${buildPrefix}" "${artifactPrefix}" artifactPosition)
    if(NOT artifactPosition EQUAL 0)
        message(FATAL_ERROR "Configure the host compiler inside UDK !artifacts")
    endif()
endif()
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "XamlCompiler must be built for the host")
endif()
add_executable(udk_xaml_compiler "${CMAKE_CURRENT_LIST_DIR}/../../../Sources/XamlCompiler/Main.cpp")
set_target_properties(udk_xaml_compiler PROPERTIES OUTPUT_NAME XamlCompiler)
target_compile_features(udk_xaml_compiler PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(udk_xaml_compiler PRIVATE /utf-8 /EHsc)
endif()

set(android_framework_sources "${CMAKE_CURRENT_LIST_DIR}")
set(android_framework_includes "${CMAKE_CURRENT_LIST_DIR}/../../../include")
if(NOT EXISTS "${android_framework_sources}/InputDispatcher.cpp")
    get_filename_component(android_framework_repository "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
    set(android_framework_sources "${android_framework_repository}/Sources/Android/AndroidBuildTools/ApplicationFramework")
    set(android_framework_includes "${android_framework_repository}/Includes/Android/AndroidBuildTools")
endif()
add_library(android_application_framework STATIC "${android_framework_sources}/InputDispatcher.cpp")
add_library(AndroidApplication::Framework ALIAS android_application_framework)
target_include_directories(android_application_framework PUBLIC "${android_framework_includes}")
target_include_directories(android_application_framework PRIVATE
    "${android_framework_includes}/UniversalDevKit/Android/AndroidBuildTools/ApplicationFramework")
target_compile_features(android_application_framework PUBLIC cxx_std_20)
target_link_libraries(android_application_framework PUBLIC UDK::XamlRuntime)
set_target_properties(android_application_framework PROPERTIES POSITION_INDEPENDENT_CODE ON)
if (MSVC)
    target_compile_options(android_application_framework PRIVATE /utf-8)
endif()

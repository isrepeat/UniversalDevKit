function(udkAddXamlCompiler root)
    set(PROJECT_SOURCE_DIR "${root}")
    set(componentPath "${root}/Projects/Components/Application/XamlCompiler.json")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${componentPath}")
    file(READ "${componentPath}" component)
    udkResolveComponentFiles("${component}" component)

    set(files)
    foreach(field IN ITEMS headers sources)
        string(JSON count LENGTH "${component}" ${field})
        if(count GREATER 0)
            math(EXPR last "${count} - 1")
            foreach(index RANGE ${last})
                string(JSON path GET "${component}" ${field} ${index})
                list(APPEND files "${root}/${path}")
            endforeach()
        endif()
    endforeach()

    string(JSON name GET "${component}" name)
    string(JSON standard GET "${component}" requirements cxxStandard)
    add_executable(udk_xaml_compiler ${files})
    set_target_properties(udk_xaml_compiler PROPERTIES OUTPUT_NAME "${name}")
    target_compile_features(udk_xaml_compiler PRIVATE "cxx_std_${standard}")
endfunction()

udkAddXamlCompiler("${udkRoot}")
if(MSVC)
    target_compile_options(udk_xaml_compiler PRIVATE /utf-8 /EHsc)
endif()
